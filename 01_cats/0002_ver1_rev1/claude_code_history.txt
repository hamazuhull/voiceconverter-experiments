"""Stage-wise access to the NVCNSF generator for subexperiment 03.

Unlike the subexperiment-01 wrapper, which only exposes end-to-end conversion,
this exposes the two stages separately so that the condition (F0 generation)
output can be inspected or replaced before the decoder (`synth`) runs:

    c    = content(x)            content encoder
    cond = condition(c, z)       F0 generation: fo = 10 * 2^((mu(z) + sigma(z) * s(c)) / 12)
    y    = synth(cond)           decoder: cyclic-noise source + neural filters

Utterances are processed in blocks of `sample_length` samples with the
streaming state carried from block to block, as in `infer.py`, so the causal
model sees the full past context (the subexperiment-01 wrapper restarted the
state at every block). A single whole-utterance call needs > 11 GB of GPU
memory in the decoder.
"""
import math
import pathlib
import sys

HERE = pathlib.Path(__file__).resolve().parent
# appended (not prepended) so this directory's own generate.py is not shadowed by subexp 01's
sys.path.append(str(HERE.parent / '01.end_to_end_feasibility'))
import vc  # noqa: E402  (also puts the repository root on sys.path; loads TF before JAX)
from vc import SR, CHECKPOINT_DIR, load_audio, trim_silence  # noqa: E402,F401

import numpy as np  # noqa: E402

# condition outputs kept per sample (all are upsampled to the waveform rate)
COND_KEYS = ('fo', 'vf', 'beta', 'alpha', 'fo_semitone_mu', 'fo_semitone_sigma_logvar', 'fo_semitone_std_raw')


class StageModel:
    def __init__(self, model_dir=CHECKPOINT_DIR, seed=1):
        import jax
        import jax.numpy as jnp
        import orbax.checkpoint
        from hparams import get_hparams
        from models import load_checkpoint
        self.jnp = jnp

        self.hparams = hp = get_hparams()
        self.rng = {'params': jax.random.key(hp['params_seed']), 'rng_stream': jax.random.key(seed)}
        manager = orbax.checkpoint.CheckpointManager(
            str(pathlib.Path(model_dir).absolute()),
            item_names=('g_model', 'd_model', 'd_fo_model') if hp['with_d_fo'] else ('g_model', 'd_model'),
            options=orbax.checkpoint.CheckpointManagerOptions(create=False, enable_async_checkpointing=False),
        )
        self.step = manager.latest_step()
        g_state = load_checkpoint(manager, self.rng, hp)[0]
        self.params = g_state.params
        apply = g_state.apply_fn
        self.block_length = hp['sample_length']
        self.ref_length = 2 * hp['sample_length']
        # with temperature=None, z_q is the nearest of these 512 codes (SpeakerSQVAEHead)
        self.codebook = np.asarray(
            self.params['params']['speaker_encoder']['SpeakerSQVAEHead_0']['codebook'])

        self._speaker = jax.jit(lambda p, r, rng: apply(p, r, None, method='speaker_encode', rngs=rng)[1])
        # (output, stream state); the first block (state None) and later blocks compile separately
        self._content = jax.jit(lambda p, x, st, rng: apply(
            p, x, stream_data=st, method='content_encode', rngs=rng)[0::2])
        self._condition = jax.jit(lambda p, c, z, st, rng: apply(
            p, c, z, stream_data=st, method='condition', rngs=rng))
        self._synth = jax.jit(lambda p, cond, st, rng: apply(
            p, cond, stream_data=st, source_key=jax.random.key(1), noise_key=jax.random.key(2),
            method='synth', rngs=rng)[0::2])

    def speaker(self, r):
        """Returns (z, z_q, code index) for a reference waveform (tiled / center-cropped as in subexp 01)."""
        if r.size < self.ref_length:
            r = np.tile(r, math.ceil(self.ref_length / r.size))
        start = (r.size - self.ref_length) // 2
        r = self.jnp.asarray(r[start:start + self.ref_length], dtype=self.jnp.float32)[None, :, None]
        aux = self._speaker(self.params, r, self.rng)
        return aux['z'], aux['z_q'], int(np.argmax(np.asarray(aux['logits']).reshape(-1)))

    def condition(self, x, z):
        """Content encoder + condition, block-wise with streaming state.

        Returns a list of per-block cond dicts (input to `synth`)."""
        jnp = self.jnp
        n = math.ceil(x.size / self.block_length)
        xp = np.pad(x, (0, n * self.block_length - x.size)).reshape(n, self.block_length)
        conds, st_c, st_d = [], None, None
        for b in xp:
            c, st_c = self._content(self.params, jnp.asarray(b, dtype=jnp.float32)[None, :, None], st_c, self.rng)
            cond, st_d = self._condition(self.params, c, z, st_d, self.rng)
            conds.append(cond)
        return conds

    def synth(self, conds, n):
        ys, st = [], None
        for cond in conds:
            y, st = self._synth(self.params, cond, st, self.rng)
            ys.append(np.asarray(y, dtype=np.float32).reshape(-1))
        return np.concatenate(ys)[:n]

    def replace(self, conds, n, **arrays):
        """Copy of block-wise `conds` with entries replaced by per-sample arrays of length n."""
        jnp = self.jnp
        L = self.block_length
        out = [dict(c) for c in conds]
        for k, v in arrays.items():
            v = np.asarray(v, dtype=np.float32)[:n]
            v = np.pad(v, (0, len(conds) * L - v.size), mode='edge')
            for i, c in enumerate(out):
                c[k] = jnp.asarray(v[i * L:(i + 1) * L]).reshape(conds[i][k].shape)
        return out

    @staticmethod
    def numpy_cond(conds, n):
        return {k: np.concatenate([np.asarray(c[k], dtype=np.float32)[0, :, 0] for c in conds])[:n]
                for k in COND_KEYS}

def select_jvs(n_per_gender):

    with open(JVS_DIR / 'gender_f0range.txt') as f:

def truth_f0(x, is_cat):
    """pYIN F0 (Hz, NaN when unvoiced) and the mask of frames trusted as ground truth."""
    f0, voiced, prob = librosa.pyin(x, **PYIN)
    mask = voiced & (prob > 0.5) & np.isfinite(f0)
    if is_cat:
        mask &= np.nan_to_num(f0) >= CAT_MIN_F0
    return f0, voiced, prob, mask


def frames_to_samples(f0, mask, n):
    """Per-sample oracle F0 (log-interpolated across untrusted frames) and voicing flag."""
    t_frame = np.arange(f0.size) * HOP
    t = np.arange(n)
    if mask.any():
        lf = np.interp(t, t_frame[mask], np.log(f0[mask]))
        fo = np.exp(lf)
    else:
        fo = np.full(n, 100.0)
    vf = np.interp(t, t_frame, mask.astype(float)) > 0.5
    return fo.astype(np.float32), vf.astype(np.float32)

# 03. Acoustic similarity analysis: where is cat F0 lost?

## Objective

Subexperiment 01 found that human→cat outputs keep a human pitch: about 228 Hz against cat targets of about 546 Hz,
an error of about 16 semitones. This subexperiment finds **which stage loses the cat F0**:

- **F0 generation** (`Condition`): the model predicts
  `fo = 10 · 2^((μ(z) + σ(z) · s(c)) / 12)`, clipped to 27.5–880 Hz. The speaker code `z` sets the mean μ and the
  spread σ. The content features `c` set the normalized contour `s` (bounded to ±3 by tanh).
- **Decoder** (`synth`): a cyclic-noise source driven by `fo`, followed by neural filters conditioned on the
  same features.

It also compares voicing, aperiodicity (decay rate β, spectral flatness) and the speaker-code quantization of
real cats against real humans.

## Method

### Self-reconstruction design

Every clip is reconstructed with **its own** speaker code. Input and output are then time-aligned, so the input's
pYIN F0 serves as frame-level ground truth for every stage. Four settings per clip (`generate.py`):

| setting | speaker code | F0 / voicing given to the decoder | isolates |
|---|---|---|---|
| `zq` | `z_q` (normal) | model | normal operation |
| `z` | continuous `z` | model | effect of the 512-code quantizer |
| `oracle_fo` | `z_q` | **true F0**; model voicing | decoder alone |
| `oracle_fo_vf` | `z_q` | **true F0 and true voicing** | decoder alone |

**Supplemental pitch sweep** (`generate_sweep.py`). The oracle runs inject cat F0 under cat conditioning, so a
decoder failure could come from the frequency or from the cat code. To separate these, each clip is re-synthesized
with its true F0 and voicing, but F0 shifted by k semitones: humans by k = 0…+24 (into the cat range) and cats by
k = 0…−24 (into the human range). If accuracy depends only on the injected frequency, both groups fall on one
curve.

### Data

| group | n | source |
|---|---|---|
| cat | 16 | meow-10k, screened from 300 random clips as in subexp 01 (top pYIN voiced ratio, median F0 250–1500 Hz); silence-trimmed, ≤ 8 s |
| human closed-set | 8 (4 M, 4 F) | jvs005, 006, 009, 011 (M) and jvs007, 008, 010, 014 (F), `VOICEACTRESS100_001` (training speakers; disjoint from the subexp-01 references) |
| human open-set | 8 (4 M, 4 F) | Common Voice ja test, the same 8 sources as subexp 01 |

### Processing and measures

- The model runs block-wise **with the streaming state carried across blocks**, as in `infer.py`. Subexp 01
  restarted the state each block, and a single whole-utterance call needs more than 11 GB of GPU memory.
- Ground truth comes from pYIN (50–2000 Hz, 10 ms hop). "Truth frames" are frames where pYIN says voiced with
  probability > 0.5; for cats the F0 must also be ≥ 200 Hz, which removes background noise (see subexp 01).
- Output F0 uses pYIN's voiced decision. Outputs are shifted back by the decoder latency, estimated per clip
  (0–5 frames, mean 1.6–2.5 frames).
- **Accuracy** is the fraction of truth frames whose pitch is within ±1 semitone. An unvoiced output frame counts as
  wrong (the pitch was not rendered). Octave errors are reported separately.

### 2. Why generation fails (`f0_along_pipeline.png`, `contour_examples.png`)

| component | cat | human closed | human open |
|---|---|---|---|
| true median F0 [Hz] | 572 | 228 | 168 |
| speaker mean μ(z_q) [Hz] | 243 | 191 | 175 |
| mean offset: μ − true [st] | **−14.5** | −3.2 | +1.2 |
| contour r (predicted vs true st) | **0.04** | 0.86 | 0.60 |
| range ratio (std predicted / std true) | 0.89 | 0.52 | 1.04 |
| frames clipped at 880 Hz | 0 | 0 | 0 |
| voicing (vf > 0.5) on truth frames | 0.99 | 0.98 | 0.97 |

- **The speaker mean μ is the main error (−14.5 st).** For cats, μ stays at 144–323 Hz whatever the true F0
  (325–1237 Hz). For humans it follows the true F0 (left panel of `f0_along_pipeline.png`).
- **The quantizer is not the cause.** Continuous `z` gives the same μ (235 Hz, offset −14.9 st). The cat
  embeddings are out of distribution (distance from `z` to the nearest code: 3.7 vs 2.0–2.3 for humans), yet 11 of
  16 cats are mapped to codes that human speakers also use (218, 280, 306, 322, 323). The μ head never outputs a
  cat-like mean, with or without the quantizer.
- **The contour also fails.** Predicted and true pitch movements are uncorrelated for cats (r = 0.04, vs 0.86 for
  jvs). The contour s comes only from the content encoder, which explains why r is identical for `z` and `z_q` (μ
  and σ only shift and scale it). A cat's pitch glide is not carried by the content features.
- The 880 Hz clip is **not** the active limit (0 % of frames clipped), because the generated F0 never gets near
  it. Cat truth frames above 880 Hz (7 % per clip on average, 9.6 % of pooled frames) would be clipped even with a
  perfect μ.
- Voicing is fine: the model marks 99 % of cat voiced frames as voiced.

### 3. Why the decoder fails on correct cat F0 (`oracle_by_f0_band.csv`, `decoder_pitch_sweep.png`, `frame_f0_histograms.png`)

Oracle F0 + voicing, cat frames by true F0:

| true F0 | frames | rendered within ±1 st | output voiced | correct when voiced |
|---|---|---|---|---|
| 200–440 Hz | 1096 | **0.93** | 0.96 | 0.97 |
| 440–880 Hz | 4284 | 0.22 | 0.41 | 0.54 (21 % octave errors) |
| 880–2000 Hz | 569 | **0.00** | 0.00 | — |

Pitch sweep, accuracy among voiced output frames, by injected F0 band:

| injected F0 | cat conditioning | human conditioning |
|---|---|---|
| 283–400 Hz | 0.92 | 0.90 |
| 400–566 Hz | 0.82 | 0.42 |
| 566–800 Hz | **0.01** | **0.00** |
| 800–1131 Hz | output never voiced | voiced, 0.00 correct |
| 1131–1600 Hz | output never voiced | voiced, 0.00 correct |

- **The decoder has a pitch ceiling around 550 Hz, and it does not depend on cat conditioning.** Human
  conditioning pushed upward fails in the same band as cat conditioning (0.00 vs 0.01 at 566–800 Hz), while cat
  conditioning pulled down into 283–566 Hz renders correctly (0.82–0.92). So the failure is about the frequency, not
  the cat timbre or content.
- Above the ceiling the decoder does not simply clip. Between 550 and 880 Hz the output collapses to an unrelated
  pitch around 330–400 Hz (e.g. cat01 in `contour_examples.png`). Above 880 Hz, cat-conditioned output contains no
  voiced pitch at all.
- The ceiling matches the training data: jvs per-speaker maximum F0 is ≤ 480 Hz (female median 410 Hz,
  `gender_f0range.txt`).
- There is also a lower limit: below about 140 Hz, 64–70 % of voiced frames are rendered an octave too low, in both
  groups. This is a separate problem that already affects human male voices (e.g. jvsM01: condition 113 Hz, output
  58 Hz).

### 4. Aperiodicity

| | cat | human closed | human open |
|---|---|---|---|
| model decay rate β (median) | 0.254 | 0.257 | 0.259 |
| log10 spectral flatness, input (voiced frames) | −10.7 | −9.0 | −9.9 |
| log10 spectral flatness, reconstruction `zq` | −7.6 | −8.6 | −8.6 |
| log10 spectral flatness, reconstruction oracle | −6.6 | −8.4 | −8.3 |

- β barely changes from speaker to speaker (0.254–0.259), so the β head does not tell cats from humans.
- Real cat calls are more tonal than human speech (flatness −10.7 vs −9.0/−9.9). Reconstructions of cats are noisier
  than humans' (−7.6 vs −8.6): the cat's tonality is not reproduced either. It gets worse with oracle F0, because
  high-F0 frames come out unvoiced (noise).

### Conclusion

**Both stages are at fault, in a chain:**

1. **F0 generation fails first, and in normal use it is the step that produces the error.** The speaker head
   maps every cat to a human-range mean F0 (−14.5 st), and the content encoder does not carry the cat's pitch
   contour (r = 0.04). This happens with or without quantization, so it is a property of the trained μ/σ heads and
   content features, not of the codebook alone.
2. **The decoder fails as well.** Even with perfect F0, it cannot render pitch above about 550 Hz. This is a
   frequency limit, independent of the cat code, and consistent with a training range of ≤ 480 Hz. About 80 % of cat
   voiced frames (440–2000 Hz) lie above it.
3. The 880 Hz `fo_range` clip is a third, currently hidden limit. It would cut about 7–10 % of cat frames once the
   first two are fixed.

Fixing only the F0 generation (e.g. supplying the cat's F0) would therefore bring end-to-end accuracy only to about
the oracle level (0.38). The decoder also has to be trained on, or augmented with, F0 up to about 1.5 kHz.

**Compared with conventional VC:** for human speakers, closed-set (jvs) generation has a small mean offset
(−3.2 st) and a good contour (r = 0.86). Open-set (CV) speakers have a good mean (+1.2 st) and a weaker contour
(r = 0.60). The cat condition is not just "more open-set": both its mean (−14.5 st) and its contour (0.04) fall
far outside both baselines, and its F0 range is outside what the decoder can render, which no human condition
reaches.

## Limitations

- 16 cats and 16 humans with one seed. Frame counts are pooled in the band tables, so clips with more voiced frames
  weigh more.
- Ground-truth F0 for noisy cat field recordings comes from pYIN with a 200 Hz floor; residual tracking errors
  remain.
- Absolute human accuracies are modest. Part of this is the decoder's low-F0 octave errors and the strict ±1 st
  criterion. The cat vs human comparisons are made under identical criteria.
- The training data is built offline from jvs as 15 identities × 11 pitch variants per speaker
  (`dataset_builders.py`). The pitch-shift range of that augmentation is not recorded in this repository. The
  ≤ 480 Hz figure is for the original jvs speakers, and the empirical decoder ceiling (~550 Hz) suggests the
  augmentation does not reach far beyond it. **This should be verified.**
- Spectral features beyond F0, voicing and flatness (formants, LTAS) were covered in subexp 01 and are not
  repeated here.

# 01. End-to-end feasibility test

## Objective

Check that the pretrained NVCNSF model can run human → cat conversion end to end, and measure how close the
generated "cat-like" sounds get to the target cat acoustically. The comparison point is conventional human → human
VC with the same model and the same sources:

| condition | target reference | why it is a reference point |
|---|---|---|
| `human2cat` | real cat vocalization (meow-10k) | the condition under study |
| `open_intra` / `open_inter` | unseen Common Voice speaker, same / other gender | open-set VC, the realistic deployment case (cats are open-set too) |
| `closed_intra` / `closed_inter` | jvs training speaker, same / other gender | closed-set VC, upper bound of what the model does well |

Concrete questions:
1. Does the pipeline produce valid, finite audio for cat references? (feasibility)
2. How much of the source→target acoustic distance does a conversion close, and how does human→cat compare with
   the conventional settings?
3. Does the output pitch follow the target into the cat F0 range?

### Model versions

| | v1 (previous run) | **v2 (this report)** |
|---|---|---|
| model | NVCNSF, `~/Documents/Repos/nvcnsf_app/app_py/checkpoints/`, step 500000 | **NVCNSFV2**, `/path/to/checkpoint/`, step 100000 |
| F0 path | F0 generated from content features + speaker code, `fo_range` 27.5–880 Hz | F0 **extracted from the input**, mapped to the target with speaker semitone statistics μ(z), σ(z), `fo_range` 27.5–1760 Hz |
| pitch control | none | decoder-side `semitone_shift` |
| human→cat F0 | whatever the model produced | **per-pair shift matched to the cat reference** (see Method) |

The v1 outputs are kept in `results_v1_old/` (summary, metrics, figures, manifest). Every v1 number in this report
comes from there.

## Data

Unchanged from v1 (same seed, same selection):

| role | source | selection (seed 0) |
|---|---|---|
| sources | Common Voice ja `test.tsv` | 4 male + 4 female speakers (`gender` labeled), one 3–7 s utterance each |
| open-set refs | Common Voice ja `test.tsv` | 2 M + 2 F speakers disjoint from the sources; up to 6 clips concatenated (≥ 4 s after silence removal) |
| closed-set refs | jvs `nonpara30` | jvs001 (M), jvs003 (M), jvs002 (F), jvs004 (F); 3 utterances concatenated |
| cat refs | meow-10k `audios/` (16 kHz) | 120 random clips screened with pYIN (150–2000 Hz); top 6 by voiced ratio with median F0 in 250–1500 Hz |

The cat screening step exists because many meow-10k clips are behaviour recordings, not vocalizations. A first
ranking by voiced ratio alone picked clips whose "F0" was pinned at the search floor (purrs / low-frequency
background noise detected as voiced). The 250–1500 Hz median-F0 window removes those. Screening results:
`data/cat_candidates.csv`.

Selected cats: `cat00` fs_cat_class_00319, `cat01` tz_cat_class_00415, `cat02` fs_cat_class_00201,
`cat03` tz_cat_class_00419, `cat04` tz_cat_class_00204, `cat05` fs_cat_class_00705.

Total: 8 sources × 14 references = **112 conversions** (48 human→cat, 16 per baseline condition).

## Method

`generate.py` runs the `infer.py` pipeline through the wrapper in `vc.py`. For each block of `sample_length` samples,
the streaming state is carried across blocks and the steps are:
1. Encode the content, extract F0 from the input, and predict the source parameters (vf, α, β).
2. Map F0 from the source's semitone statistics to the target's (`map_fo`), then apply `semitone_shift`.
3. Compute the mgc and decode.

References are silence-trimmed as in `infer.py`, then tiled / center-cropped to 2 × `sample_length` (2.73 s). The
source utterance is processed the same way to obtain its own statistics. The quantized speaker code `z_q` drives the
conversion.

**Cat-matched pitch shift (human→cat only).** Each pair is first converted with shift 0; that output is kept in
`data/conv_s0/`. The shift is then `12·log2(F0_cat / F0_out)`, where both F0s are pYIN medians (150–2000 Hz, the
cat-screening settings), clipped to ±24 st. The conversion is repeated with that shift and saved to `data/conv/`.
The shift and target are logged in `data/manifest.csv` (`shift_st`, `target_f0_hz`). Human baselines use shift 0,
so they measure the model's own F0 mapping.

`analyze.py` computes, on 16 kHz versions of every file (common bandwidth: the cat recordings are 16 kHz):

| metric | definition |
|---|---|
| `mfcc` | Euclidean distance of utterance-mean MFCC c1–c24 over non-silent frames (librosa units) |
| `ltas` | RMS difference (dB) of level-normalized long-term average spectra, 100 Hz–7 kHz |
| `spk` | cosine similarity of the model's own continuous speaker embedding `z` |
| `f0` | pYIN median F0 of voiced frames (60–1600 Hz); cat targets use the 150–2000 Hz estimate (see limitations) |
| `gap_closed` | `1 − d(conv, ref) / d(src, ref)`: 1 = reached the target, 0 = did not move, < 0 = moved away |

To give the distances a scale, `reference_spread.csv` contains the distances among the real references
themselves (cat–cat, human–human, cat–human).

### Findings

1. **The pipeline runs end to end with the new model.** All 112 conversions finished with finite output and the
   expected length, as in v1.
2. **Pitch now reaches the cat range (v1's main failure is fixed).** Output F0 follows the cat target from 280 to
   1230 Hz with a mean error of 1.3 st (median 0.2 st). In v1 the output stayed at 65–400 Hz with a 16 st error.
   cat00 (≈ 1.2 kHz) was impossible in v1 because of the 880 Hz `fo_range` cap. It is now reproduced at 1220 Hz and
   is also the cat with the largest spectral gain (0.53). Two things made this possible: the decoder renders F0 up
   to ≥ 1.9 kHz (subexp 03) and the semitone shift moves the F0 there. Without the shift, μ(z) for cats is too low
   (subexp 03), so the shift is necessary. 43 of 48 pairs end within 3 st of the target. The exceptions are mostly
   cat01: its median-based shift undershoots because the cat01 reference contains low non-meow components, so its
   output stays at 166–351 Hz.
3. **Human→cat now moves toward the target about as much as conventional VC, but the remaining gap is large.**
   - Human→cat closes 23 % of the MFCC gap (v1 15 %) and 16 % of the LTAS gap (v1 10 %). This is within the range
     of the v2 human baselines (16–29 % MFCC, 9–16 % LTAS). In v1 it was below every baseline.
   - In absolute terms the output is still 47.1 from its cat target. That is below the cat–human spread (55.3) but
     far from the 30–32 the human conditions reach. Because the starting gap is about 1.5× larger (62.8 vs 38–45),
     an equal fraction leaves much more distance.
   - The outputs also moved away from the source: d(conv, src) went from 34.0 to 51.5.
4. **The spectra show what pitch changed and what it did not** (`ltas.png`, `spectrogram_examples.png`).
   - Changed: the output LTAS lost the human low-frequency (< 300 Hz) peak, gained a bump around 1–1.5 kHz, and
     follows the cats' falling slope below 500 Hz.
   - Not changed: it still lacks most of the cats' 0.8–3 kHz energy. In the spectrogram the cat00 output has a
     sustained ≈ 1.2 kHz harmonic series like the cat, but it keeps the source's syllable timing and its broadband
     consonant noise up to 8 kHz. The cat's long, smooth tonal calls with gliding pitch are not produced.
   - The outputs are human utterances at cat pitch, not cat calls.
5. **Voicing moved toward the cat.** The output voiced ratio is 0.64 (v1 0.44; cat references 0.75).
6. **The model's own speaker embedding** rates the outputs closer to the cat (cosine 0.54, v1 0.37). That is still
   below the v2 cat–cat similarity (0.72). Because the encoder that judges the output is the one that conditioned it,
   this is optimistic and is not taken as evidence of cat-likeness. Subexperiment 2 tests this with an independent
   classifier.

**Compared with conventional VC:**
- In v1, the closed-set inter-gender condition closed 39 % of the MFCC gap with a 2 st F0 error, while human→cat
  closed 15 % with a 16 st error, so the species gap looked out of reach.
- In v2, with pitch matched, human→cat (23 %) is level with open-set inter-gender (23 %) and closed-set intra-gender
  (22 %) conversion. Its remaining distance (47 vs 30–32) is still far larger, and the residual is spectral and
  temporal (timbre, call shape), not pitch.
- The human baselines themselves got worse in v2 (gap closed −0.05 to −0.10, F0 error 4.5–5.7 st). So part of the
  "catching up" is the baselines dropping, not only human→cat improving. Open-set is still worse than closed-set
  (0.16–0.23 vs 0.22–0.29).

## Limitations

- Small sample (8 sources, 6 cats), single seed. This is a feasibility check, not a statistical comparison; no
  significance tests are reported.
- **The pitch shift is oracle-informed.** It is computed from the target cat's measured F0, so v2 human→cat is an
  upper bound for pitch control, not a fully automatic conversion. The shift-0 outputs are saved in `data/conv_s0/`,
  but their metrics have not been computed. The contributions of the new model and of the shift are therefore not
  separated.
- v1 and v2 differ in checkpoint, architecture and training. Changes in the human baselines show that absolute
  numbers are not strictly comparable. The comparisons above are within-version (human→cat vs baselines) or
  directional.
- Cat clips are 16 kHz field recordings with background noise. For one cat (cat05), the wide-range pYIN locks onto
  low-frequency background at 157 Hz (vs 280 Hz in the 150–2000 Hz band). This is why cat target F0 uses the
  meow-band estimate. Both values are kept in `metrics.csv` (`f0_ref`, `f0_ref_meow`).
- Cat clips can mix meows with other sounds (purr, noise), and the 2.73 s reference crop may include them. This is
  the likely cause of cat01's poor shift.
- Utterance-mean MFCC and LTAS ignore temporal structure (a cat call's pitch contour, duration), which
  subexperiment 3 covers.
- The speaker-embedding cosine is noisy (values spanning −0.8 to 1), so it is reported only as a raw similarity.

# 03. Acoustic similarity analysis: where is cat F0 lost?

## Objective

With the v1 model, subexperiment 01 found that human→cat outputs kept a human pitch: about 228 Hz against cat
targets of about 546 Hz, an error of about 16 semitones. This subexperiment finds **which stage of the model loses
the cat F0**, and checks whether the pitch-controllable v2 model fixes it. It also compares voicing, aperiodicity
(decay rate β, spectral flatness) and the speaker-code quantization of real cats against real humans.

### Model versions and their F0 stages

| | v1 (previous run) | **v2 (this report)** |
|---|---|---|
| checkpoint | `~/Documents/Repos/nvcnsf_app/app_py/checkpoints/`, step 500000 | `/path/to/checkpoint/`, step 100000 |
| F0 stage | **F0 generation** (`Condition`): `fo = 10 · 2^((μ(z) + σ(z) · s(c)) / 12)`, where the contour `s` comes from the content features | **F0 extractor**: reads F0 from the input waveform; the speaker code only gives semitone statistics μ(z), σ(z) for mapping to a target speaker (`map_fo`), plus an optional `semitone_shift` |
| F0 range (`fo_range`) | 27.5–880 Hz | 27.5–1760 Hz |
| decoder | cyclic-noise source driven by `fo` + neural filters | same structure (`decode`), retrained |

In self-reconstruction (source code = target code) the v2 mapping is the identity, so the F0 stage does not depend
on `z`. The output column names are kept from v1: `gen` is the F0 stage (now the extractor) and `mu` is μ(z). The v1
outputs are kept in `results_v1_old/`, and every v1 number below comes from there.

## Method

### Self-reconstruction design

Every clip is reconstructed with **its own** speaker code. Input and output are then time-aligned, so the input's
pYIN F0 serves as frame-level ground truth for every stage. Four settings per clip (`generate.py`):

| setting | speaker code | F0 / voicing given to the decoder | isolates |
|---|---|---|---|
| `zq` | `z_q` (normal) | model | normal operation |
| `z` | continuous `z` | model | effect of the speaker-code quantizer (v2: no effect on F0 by construction) |
| `oracle_fo` | `z_q` | **true F0**; model voicing | decoder alone |
| `oracle_fo_vf` | `z_q` | **true F0 and true voicing** | decoder alone |

**Supplemental pitch sweep** (`generate_sweep.py`). The oracle runs inject cat F0 under cat conditioning, so a
decoder failure could come from the frequency or from the cat code. To separate these, each clip is re-synthesized
with its true F0 and voicing, but F0 shifted by k semitones: humans by k = 0…+24 (into the cat range) and cats by
k = 0…−24 (into the human range). If accuracy depends only on the injected frequency, both groups fall on one
curve.

### Data

Unchanged from v1 (same seed, same clips):

| group | n | source |
|---|---|---|
| cat | 16 | meow-10k, screened from 300 random clips as in subexp 01 (top pYIN voiced ratio, median F0 250–1500 Hz); silence-trimmed, ≤ 8 s |
| human closed-set | 8 (4 M, 4 F) | jvs005, 006, 009, 011 (M) and jvs007, 008, 010, 014 (F), `VOICEACTRESS100_001` (training speakers; disjoint from the subexp-01 references) |
| human open-set | 8 (4 M, 4 F) | Common Voice ja test, the same 8 sources as subexp 01 |

### Processing and measures

- `model.py` runs the v2 stages block-wise **with the streaming state carried across blocks**, as in `infer.py`. A
  single whole-utterance call needs more than 11 GB of GPU memory. The stages are content encoder, F0 extractor,
  semitone statistics and `map_fo`, source parameters (vf, α, β), mgc, and decode. The decoder inputs can be
  replaced for the oracle settings.
- Ground truth comes from pYIN (50–2000 Hz, 10 ms hop). "Truth frames" are frames where pYIN says voiced with
  probability > 0.5; for cats the F0 must also be ≥ 200 Hz, which removes background noise (see subexp 01).
- Output F0 uses pYIN's voiced decision. Outputs are shifted back by the decoder latency, estimated per clip
  (mean 1.9–2.1 frames).
- **Accuracy** is the fraction of truth frames whose pitch is within ±1 semitone. An unvoiced output frame counts as
  wrong (the pitch was not rendered). Octave errors are reported separately.

## Results

### 1. Accuracy per stage

These are means over clips (`results/summary.csv`, `stage_accuracy.png`).

| stage | cat v2 | human closed v2 | human open v2 | cat v1 | human closed v1 | human open v1 |
|---|---|---|---|---|---|---|
| F0 stage vs true F0 (v1: generation, v2: extractor) | **0.85** | 0.93 | 0.94 | 0.13 | 0.28 | 0.21 |
| decoder: output vs given F0 | 0.95 | 0.93 | 0.94 | 0.72 | 0.70 | 0.54 |
| decoder with oracle F0 | 0.99 | 0.99 | 0.99 | — | — | — |
| decoder with oracle F0 + voicing | **0.995** | 0.97 | 0.93 | 0.38 | 0.59 | 0.46 |
| end to end: output vs true F0 | **0.84** | 0.95 | 0.95 | 0.12 | 0.19 | 0.17 |

**Both stages that failed in v1 now work.** End-to-end cat pitch accuracy rose from 0.12 to 0.84. Human accuracy
also rose (0.17–0.19 → 0.95), so the old pipeline was weak for everyone and especially for cats. Cats still trail
humans by about 0.1, and that loss is in the F0 extractor. The decoder renders what it is given (0.95) and renders
the true cat F0 almost perfectly (0.995).

### 2. The F0 stage (`f0_along_pipeline.png`, `contour_examples.png`)

| component | cat v2 | human closed v2 | human open v2 | cat v1 |
|---|---|---|---|---|
| true median F0 [Hz] | 572 | 228 | 168 | 572 |
| F0-stage median F0 [Hz] | 576 | 229 | 169 | ≈ 243 |
| contour r (F0 stage vs true, st) | **0.76** | 0.98 | 0.92 | 0.04 |
| range ratio (std F0 stage / std true) | 1.45 | 1.06 | 1.19 | 0.89 |
| frames at the F0 cap | 0 (cap 1760 Hz) | 0 | 0 | 0 (cap 880 Hz) |
| voicing (vf > 0.5) on truth frames | 0.97 | 0.97 | 0.95 | 0.99 |

- **The cat contour is now carried** (r = 0.76, v1 0.04). In v1 the contour came from content features that did not
  encode a cat's pitch glide. The v2 extractor reads it from the waveform.
- **The extractor's errors are concentrated at high pitch.** By true-F0 band (pooled cat frames,
  `oracle_by_f0_band.csv`):

  | true F0 | frames | extractor ±1 st | end to end ±1 st | decoder with oracle F0 + voicing | v1 end to end with oracle |
  |---|---|---|---|---|---|
  | 200–440 Hz | 1096 | 0.92 | 0.92 | 0.99 | 0.93 |
  | 440–880 Hz | 4284 | 0.88 | 0.88 | 1.00 | 0.22 |
  | 880–2000 Hz | 569 | **0.33** | 0.30 | 0.99 | 0.00 |

  About 10 % of cat voiced frames are above 880 Hz. Per clip, the lowest accuracies are cat05 (median 1237 Hz,
  0.29) and cat04 (819 Hz, 0.61). Two lower-pitched noisy clips are also low: cat07 (543 Hz, 0.57) and cat02
  (610 Hz, 0.72). The other 12 cats reach 0.81–0.99. The extractor's range ratio of 1.45 for cats means its contour
  swings more widely than the true one; this is consistent with jumps at high pitch.
- **The speaker statistics μ(z) are still too low for cats.**

  | | cat v2 | human closed v2 | human open v2 | cat v1 |
  |---|---|---|---|---|
  | μ(z) [Hz] | 367 | 208 | 193 | 243 |
  | offset from true median [st] | **−7.7** | −2.3 | +3.0 | −14.5 |

  The bias halved from v1, but it is still about 3× the human offsets. It does not affect self-reconstruction (the
  mapping is the identity). It does affect conversion *to* a cat: `map_fo` would put human speech about 8 st below
  the cat's real pitch. This is why subexp 01 v2 needed a per-pair `semitone_shift` (+9 st on average).
- **Quantization.** The continuous `z` gives the same μ, so the quantizer is not the cause. The 16 cats use only 4
  codes (119, 276, 359, 455), and two of these (359, 455) are also used by human speakers. The distance from `z` to
  the nearest code is larger for cats (7.3 vs 5.8 for humans), so cat embeddings are still out of distribution.

### 3. The decoder (`decoder_pitch_sweep.png`, `sweep_by_f0_band.csv`, `frame_f0_histograms.png`)

Pitch sweep, accuracy (±1 st) by injected F0 band; v1 values are accuracy among voiced output frames:

| injected F0 | cat conditioning v2 | human conditioning v2 | cat v1 | human v1 |
|---|---|---|---|---|
| 100–141 Hz | 0.99 | 0.93 | — | octave errors (64–70 %) |
| 283–400 Hz | 0.99 | 0.98 | 0.92 | 0.90 |
| 400–566 Hz | 1.00 | 0.98 | 0.82 | 0.42 |
| 566–800 Hz | 1.00 | 0.97 | **0.01** | **0.00** |
| 800–1131 Hz | 0.99 | 0.97 | never voiced | 0.00 |
| 1131–1600 Hz | 1.00 | 0.97 | never voiced | 0.00 |
| 1600–2263 Hz | 1.00 (50 frames) | 1.00 (150 frames) | — | — |

- **The v1 decoder pitch ceiling (~550 Hz) is gone.** v2 renders injected F0 from 100 Hz to about 2 kHz with
  ≥ 0.93 accuracy and < 1 % octave errors, under both cat and human conditioning. v1 collapsed above 566 Hz and was
  silent above 880 Hz under cat conditioning.
- **The low-pitch octave errors of v1 are also gone.** v1 rendered 64–70 % of voiced frames below 140 Hz an
  octave too low; v2 has < 1 % octave errors in every band.

### 4. Aperiodicity

| | cat v2 | human closed v2 | human open v2 | cat v1 | human v1 |
|---|---|---|---|---|---|
| model decay rate β (median) | **0.77** | 0.39 | 0.43 | 0.254 | 0.257–0.259 |
| log10 spectral flatness, input (voiced frames) | −10.7 | −9.0 | −9.9 | −10.7 | −9.0 / −9.9 |
| log10 spectral flatness, reconstruction `zq` | **−9.7** | −8.8 | −8.9 | −7.6 | −8.6 |
| log10 spectral flatness, reconstruction oracle | −9.9 | −8.8 | −8.8 | −6.6 | −8.4 / −8.3 |

- β now differs by speaker group (cats about 2× humans). In v1 it was nearly constant (0.254–0.259), so the β head
  did not tell cats from humans.
- Cat reconstructions are much more tonal than in v1 (−9.7 vs −7.6), closer to the real cat input (−10.7), and more
  tonal than human reconstructions. v1's high-F0 frames came out as noise; that no longer happens.

### Conclusion

v1 lost the cat F0 in two places. The F0 generator mapped every cat to a human-range mean (−14.5 st) with no
contour (r = 0.04). The decoder could not render pitch above about 550 Hz. A third, hidden limit was the 880 Hz
`fo_range` cap.

**v2 removes all three:**

1. F0 now comes from the input, so the cat's pitch level and contour are carried (end to end 0.12 → 0.84).
2. The decoder renders pitch from 100 Hz to about 2 kHz, independent of cat or human conditioning.
3. `fo_range` now reaches 1760 Hz, and no frames hit it.

**What remains:**

1. **The F0 extractor above 880 Hz** (0.33 accuracy on about 10 % of cat frames, mostly very high-pitched cats).
   Training or augmenting the extractor with F0 up to about 1.5 kHz would address this.
2. **μ(z) for cats (−7.7 st).** This only matters for conversion to a cat target, where it currently has to be
   corrected with `semitone_shift` (subexp 01 v2).

**Compared with conventional VC:** in v1 the cat condition was far outside both human baselines on every F0
measure: mean −14.5 st, contour 0.04, and an F0 range beyond the decoder. In v2, cat self-reconstruction (0.84) is
close to closed-set (0.95) and open-set (0.95) humans, and its contour (0.76) is below but in the same regime as
humans (0.92–0.98). The cat-specific deficits are now a narrow high-pitch extractor problem and a biased speaker
statistic, not a structural inability to represent cat pitch.

## Limitations

- 16 cats and 16 humans with one seed. Frame counts are pooled in the band tables, so clips with more voiced frames
  weigh more.
- Ground-truth F0 for noisy cat field recordings comes from pYIN with a 200 Hz floor; residual tracking errors
  remain. Some low extractor scores on mid-pitch cats (cat02, cat07) may partly reflect pYIN errors on noisy clips.
- Self-reconstruction tests pitch, not cat timbre. A correct F0 does not mean the output sounds like a cat; subexp
  01 v2 shows the spectral gap to real cats is still mostly open.
- The sweep's top band (1600–2263 Hz) has few frames (50 cat, 150 human) and lies partly above the 1760 Hz
  `fo_range`.
- v1 and v2 differ in checkpoint, architecture and training, so per-stage comparisons are directional. The two v1
  stages (F0 generation, old decoder) do not exist in v2 in the same form.
- Spectral features beyond F0, voicing and flatness (formants, LTAS) are covered in subexp 01 and are not repeated
  here.
